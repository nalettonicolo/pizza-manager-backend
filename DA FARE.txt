import { PrismaClient, Ruolo } from '@prisma/client'
import * as argon2 from 'argon2'

const prisma = new PrismaClient()

async function main() {
  const passwordHash = await argon2.hash('password123')

  const tenant = await prisma.tenant.create({
    data: {
      nome: 'Pizzeria Demo'
    }
  })

  await prisma.user.create({
    data: {
      nome: 'Admin',
      email: 'admin@pizzeria.it',
      password: passwordHash,
      ruolo: Ruolo.ADMIN,
      tenantId: tenant.id
    }
  })

  await prisma.configurazioneCosti.create({
    data: {
      tenantId: tenant.id,
      costoImpasto: 0.45,
      costoEnergia: 0.30
    }
  })

  const mozzarella = await prisma.ingrediente.create({
    data: {
      nome: 'Mozzarella',
      costo: 0.8,
      tenantId: tenant.id
    }
  })

  const pomodoro = await prisma.ingrediente.create({
    data: {
      nome: 'Pomodoro',
      costo: 0.4,
      tenantId: tenant.id
    }
  })

  const margherita = await prisma.prodotto.create({
    data: {
      nome: 'Margherita',
      prezzo: 8.0,
      tenantId: tenant.id
    }
  })

  await prisma.prodottoIngrediente.create({
    data: {
      prodottoId: margherita.id,
      ingredienteId: mozzarella.id,
      quantita: 1
    }
  })

  await prisma.prodottoIngrediente.create({
    data: {
      prodottoId: margherita.id,
      ingredienteId: pomodoro.id,
      quantita: 1
    }
  })

  console.log('✅ Seed completato')
  console.log('Email: admin@pizzeria.it')
  console.log('Password: password123')
}

main()
  .catch((e) => {
    console.error(e)
    process.exit(1)
  })
  .finally(async () => {
    prisma.$disconnect()
  })
